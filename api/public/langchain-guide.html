Using Arch Tools with LangChain

Add 58 production tools to your LangChain agents in minutes. Web scraping, AI generation, crypto data, OCR, and more — all available as LangChain tools.

Install

pip install arch-tools langchain langchain-openai

Quick Start

from arch_tools import ArchTools
from langchain.tools import Tool
from langchain.agents import initialize_agent, AgentType
from langchain_openai import ChatOpenAI

# Initialize Arch Tools
arch = ArchTools(api_key="arch_your_key_here")

# Wrap as LangChain tools
tools = [
    Tool(
        name="WebScraper",
        func=lambda url: arch.web_scrape(url)["text"],
        description="Scrape and extract text from any URL"
    ),
    Tool(
        name="CryptoPrice",
        func=lambda symbol: str(arch.crypto_price(symbol)),
        description="Get current price for a crypto symbol (e.g. bitcoin, ethereum)"
    ),
    Tool(
        name="AIGenerate",
        func=lambda prompt: arch.ai_generate(prompt)["text"],
        description="Generate text using Claude, GPT-4, or other models"
    ),
    Tool(
        name="SearchWeb",
        func=lambda query: str(arch.search_web(query)["results"][:3]),
        description="Search the web and return top results"
    ),
]

# Create agent
llm = ChatOpenAI(model="gpt-4o")
agent = initialize_agent(tools, llm, agent=AgentType.ZERO_SHOT_REACT_DESCRIPTION, verbose=True)

# Run
result = agent.run("What is Bitcoin's current price and summarize recent crypto news?")
print(result)

All 58 Tools as LangChain Tools

tools = []
tool_names = [
    ("WebScrape", "web_scrape", "Scrape text from a URL"),
    ("SearchWeb", "search_web", "Search the web"),
    ("Screenshot", "screenshot_capture", "Take a screenshot of a URL"),
    ("AIGenerate", "ai_generate", "Generate text with AI"),
    ("OCR", "ocr_extract", "Extract text from images"),
    ("CryptoPrice", "crypto_price", "Get crypto price"),
    ("EmailVerify", "email_verify", "Verify an email address"),
    ("Summarize", "summarize", "Summarize text"),
]

for name, method, desc in tool_names:
    tools.append(Tool(
        name=name,
        func=lambda x, m=method: str(getattr(arch, m)(x)),
        description=desc
    ))
